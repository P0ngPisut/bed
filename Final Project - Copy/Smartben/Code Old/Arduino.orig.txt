/*
 * Arduino Uno - Smart Bed Controller for ESP32
 * Fixed Serial Communication - Compatible with ESP32 Pin 25/26
 * Version: Compatible 1.0
 */

// === Pin Definitions ===
// Back Motor (หลัง)
#define BACK_PWM_F 3
#define BACK_PWM_R 5

// Side Motor (ตะแคง)
#define SIDE_PWM_F 6
#define SIDE_PWM_R 9

// Leg Motor (ขา)
#define LEG_PWM_F 10
#define LEG_PWM_R 11

// Other pins
#define EMERGENCY_PIN A0
#define LED_STATUS A1

// === System State ===
enum SystemState {
  STATE_READY,
  STATE_MOVING,
  STATE_EMERGENCY
} currentState = STATE_READY;

// === Position Variables ===
struct BedPosition {
  int backLevel = 0;        // 0-10 (0=flat, 10=raised)
  int sideLevel = 5;        // 0-10 (5=center)
  int legLevel = 5;         // 0-10 (5=neutral) - FLAT position
} currentPos, targetPos;

// === Control Variables ===
bool isMoving = false;
unsigned long stepTimer = 0;
unsigned long emergencyTimer = 0;
unsigned long statusTimer = 0;
unsigned long heartbeatTimer = 0;

// === Configuration ===
const int MOTOR_SPEED = 255;        // PWM speed (0-255)
const int STEP_DELAY = 2000;        // ms per step
const int EMERGENCY_CHECK = 50;     // ms
const int STATUS_INTERVAL = 3000;   // ms
const int HEARTBEAT_INTERVAL = 1000; // Send heartbeat every 1 sec

// === Serial Communication ===
String inputBuffer = "";
bool commandReady = false;

void setup() {
  // ✅ ใช้ Serial ปกติ (Pin 0,1) เพื่อสื่อสารกับ ESP32
  Serial.begin(9600);
  delay(1000);
  
  // Initialize hardware
  setupMotorPins();
  pinMode(EMERGENCY_PIN, INPUT_PULLUP);
  pinMode(LED_STATUS, OUTPUT);
  
  // Initialize system
  stopAllMotors();
  currentState = STATE_READY;
  digitalWrite(LED_STATUS, HIGH);
  
  // Set initial positions (FLAT = default)
  currentPos.backLevel = 0;   // Flat
  currentPos.sideLevel = 5;   // Center
  currentPos.legLevel = 5;    // Neutral - FLAT position
  
  targetPos.backLevel = 0;
  targetPos.sideLevel = 5;
  targetPos.legLevel = 5;
  
  // Clear serial buffer
  clearSerialBuffer();
  
  // ✅ ส่งสัญญาณเริ่มต้นไปยัง ESP32
  Serial.println("ARDUINO_READY");
  delay(100);
  Serial.println("Smart Bed Arduino Ready - Pin 0/1 Serial");
  Serial.println("Default Position: FLAT (0,5,5)");
  Serial.println("Waiting for ESP32 commands...");
  
  sendStatus();
  heartbeatTimer = millis();
}

void loop() {
  unsigned long now = millis();
  
  // Check emergency button
  if (now - emergencyTimer >= EMERGENCY_CHECK) {
    checkEmergencyButton();
    emergencyTimer = now;
  }
  
  // Handle serial communication
  handleSerialInput();
  
  // Process commands
  if (commandReady) {
    processCommand();
    commandReady = false;
  }
  
  // Handle movement
  if (isMoving && currentState == STATE_MOVING) {
    handleMovement(now);
  }
  
  // Send periodic status
  if (now - statusTimer >= STATUS_INTERVAL) {
    sendStatus();
    statusTimer = now;
  }
  
  // Send heartbeat to ESP32
  if (now - heartbeatTimer >= HEARTBEAT_INTERVAL) {
    Serial.println("HEARTBEAT");
    heartbeatTimer = now;
  }
  
  // Update LED
  updateStatusLED();
}

// === Serial Communication ===
void handleSerialInput() {
  while (Serial.available()) {
    char c = Serial.read();
    
    if (c == '\n' || c == '\r') {
      if (inputBuffer.length() > 0) {
        commandReady = true;
      }
    } else if (c >= 32 && c <= 126) {  // Printable ASCII
      inputBuffer += c;
      
      // Prevent overflow
      if (inputBuffer.length() > 50) {
        Serial.println("ERROR:BUFFER_OVERFLOW");
        clearInputBuffer();
      }
    }
  }
}

void processCommand() {
  String cmd = inputBuffer;
  cmd.trim();
  cmd.toUpperCase();
  
  // ✅ ส่ง Echo กลับไปยัง ESP32 เพื่อยืนยันการรับคำสั่ง
  Serial.print("RECEIVED: ");
  Serial.println(cmd);
  
  clearInputBuffer();
  
  if (cmd == "STATUS") {
    sendStatus();
  }
  else if (cmd.startsWith("MOVE:")) {
    processMoveCommand(cmd.substring(5));
  }
  else if (cmd.startsWith("TEST:")) {
    processTestCommand(cmd.substring(5));
  }
  else if (cmd == "STOP") {
    activateEmergencyStop();
  }
  else if (cmd == "RESET") {
    resetToFlat();
  }
  else if (cmd == "PING") {
    Serial.println("PONG");
  }
  else if (cmd == "HELLO") {
    Serial.println("ARDUINO_READY");
  }
  else if (cmd.length() > 0) {
    Serial.print("ERROR:UNKNOWN_COMMAND:");
    Serial.println(cmd);
  }
}

void processMoveCommand(String params) {
  if (currentState == STATE_EMERGENCY) {
    Serial.println("ERROR:EMERGENCY_ACTIVE");
    return;
  }
  
  if (isMoving) {
    Serial.println("ERROR:ALREADY_MOVING");
    return;
  }
  
  // Parse: "5,5,7" (backLevel, sideLevel, legLevel)
  int comma1 = params.indexOf(',');
  int comma2 = params.indexOf(',', comma1 + 1);
  
  if (comma1 == -1 || comma2 == -1) {
    Serial.println("ERROR:INVALID_FORMAT");
    return;
  }
  
  int newBackLevel = params.substring(0, comma1).toInt();
  int newSideLevel = params.substring(comma1 + 1, comma2).toInt();
  int newLegLevel = params.substring(comma2 + 1).toInt();
  
  // Validate ranges
  if (newBackLevel < 0 || newBackLevel > 10 || 
      newSideLevel < 0 || newSideLevel > 10 || 
      newLegLevel < 0 || newLegLevel > 10) {
    Serial.println("ERROR:OUT_OF_RANGE");
    return;
  }
  
  // Set targets and start movement
  targetPos.backLevel = newBackLevel;
  targetPos.sideLevel = newSideLevel;
  targetPos.legLevel = newLegLevel;
  
  startMovement();
}

void processTestCommand(String params) {
  if (currentState == STATE_EMERGENCY) {
    Serial.println("ERROR:EMERGENCY_ACTIVE");
    return;
  }
  
  Serial.print("Testing motor: ");
  Serial.println(params);
  
  if (params == "BACK_UP") {
    moveBackUp();
    delay(2000);
    stopAllMotors();
    Serial.println("Back up test complete");
  }
  else if (params == "BACK_DOWN") {
    moveBackDown();
    delay(2000);
    stopAllMotors();
    Serial.println("Back down test complete");
  }
  else if (params == "SIDE_LEFT") {
    moveSideLeft();
    delay(2000);
    stopAllMotors();
    Serial.println("Side left test complete");
  }
  else if (params == "SIDE_RIGHT") {
    moveSideRight();
    delay(2000);
    stopAllMotors();
    Serial.println("Side right test complete");
  }
  else if (params == "LEG_UP") {
    moveLegUp();
    delay(2000);
    stopAllMotors();
    Serial.println("Leg up test complete");  
  }
  else if (params == "LEG_DOWN") {
    moveLegDown();
    delay(2000);
    stopAllMotors();
    Serial.println("Leg down test complete");
  }
  else if (params == "ALL_STOP") {
    stopAllMotors();
    Serial.println("All motors stopped");
  }
  else {
    Serial.println("ERROR:UNKNOWN_TEST_COMMAND");
  }
}

void startMovement() {
  isMoving = true;
  currentState = STATE_MOVING;
  stepTimer = millis();
  
  Serial.println("MOVING");
  sendStatus();
  
  Serial.print("Movement started to: ");
  Serial.print(targetPos.backLevel); Serial.print(",");
  Serial.print(targetPos.sideLevel); Serial.print(",");
  Serial.println(targetPos.legLevel);
}

void resetToFlat() {
  if (isMoving) {
    Serial.println("ERROR:ALREADY_MOVING");
    return;
  }
  
  // Reset to FLAT position (0,5,5)
  targetPos.backLevel = 0;    // Flat
  targetPos.sideLevel = 5;    // Center  
  targetPos.legLevel = 5;     // Neutral - FLAT position
  
  startMovement();
  Serial.println("Resetting to FLAT position (0,5,5)");
}

// === Movement Control ===
void handleMovement(unsigned long now) {
  if (now - stepTimer < STEP_DELAY) {
    return;
  }
  
  stopAllMotors();
  bool movementComplete = true;
  
  // Sequential movement: Back → Side → Leg
  
  // Back level control
  if (currentPos.backLevel != targetPos.backLevel) {
    if (currentPos.backLevel < targetPos.backLevel) {
      moveBackUp();
      currentPos.backLevel++;
    } else {
      moveBackDown();
      currentPos.backLevel--;
    }
    Serial.print("Back: Level "); Serial.println(currentPos.backLevel);
    movementComplete = false;
  }
  // Side level control
  else if (currentPos.sideLevel != targetPos.sideLevel) {
    if (currentPos.sideLevel < targetPos.sideLevel) {
      moveSideRight();
      currentPos.sideLevel++;
    } else {
      moveSideLeft();
      currentPos.sideLevel--;
    }
    Serial.print("Side: Level "); Serial.println(currentPos.sideLevel);
    movementComplete = false;
  }
  // Leg level control
  else if (currentPos.legLevel != targetPos.legLevel) {
    if (currentPos.legLevel < targetPos.legLevel) {
      moveLegUp();
      currentPos.legLevel++;
    } else {
      moveLegDown();
      currentPos.legLevel--;
    }
    Serial.print("Leg: Level "); Serial.println(currentPos.legLevel);
    movementComplete = false;
  }
  
  if (movementComplete) {
    completeMovement();
  } else {
    stepTimer = now;
  }
}

void completeMovement() {
  isMoving = false;
  currentState = STATE_READY;
  stopAllMotors();
  
  Serial.println("COMPLETE");
  sendStatus();
  
  Serial.println("Movement completed successfully");
}

// === Motor Control Functions ===
void setupMotorPins() {
  // Back motor PWM pins
  pinMode(BACK_PWM_F, OUTPUT);
  pinMode(BACK_PWM_R, OUTPUT);
  
  // Side motor PWM pins
  pinMode(SIDE_PWM_F, OUTPUT);
  pinMode(SIDE_PWM_R, OUTPUT);
  
  // Leg motor PWM pins
  pinMode(LEG_PWM_F, OUTPUT);
  pinMode(LEG_PWM_R, OUTPUT);
}

void moveBackUp() {
  analogWrite(BACK_PWM_F, MOTOR_SPEED);
  analogWrite(BACK_PWM_R, 0);
}

void moveBackDown() {
  analogWrite(BACK_PWM_F, 0);
  analogWrite(BACK_PWM_R, MOTOR_SPEED);
}

void moveSideLeft() {
  analogWrite(SIDE_PWM_F, MOTOR_SPEED);
  analogWrite(SIDE_PWM_R, 0);
}

void moveSideRight() {
  analogWrite(SIDE_PWM_F, 0);
  analogWrite(SIDE_PWM_R, MOTOR_SPEED);
}

void moveLegUp() {
  analogWrite(LEG_PWM_F, MOTOR_SPEED);
  analogWrite(LEG_PWM_R, 0);
}

void moveLegDown() {
  analogWrite(LEG_PWM_F, 0);
  analogWrite(LEG_PWM_R, MOTOR_SPEED);
}

void stopAllMotors() {
  // Stop all motors by setting PWM to 0
  analogWrite(BACK_PWM_F, 0);
  analogWrite(BACK_PWM_R, 0);
  analogWrite(SIDE_PWM_F, 0);
  analogWrite(SIDE_PWM_R, 0);
  analogWrite(LEG_PWM_F, 0);
  analogWrite(LEG_PWM_R, 0);
}

// === Safety Functions ===
void checkEmergencyButton() {
  static bool lastButtonState = HIGH;
  bool currentState = digitalRead(EMERGENCY_PIN);
  
  if (lastButtonState == HIGH && currentState == LOW) {
    activateEmergencyStop();
  }
  
  lastButtonState = currentState;
}

void activateEmergencyStop() {
  isMoving = false;
  currentState = STATE_EMERGENCY;
  stopAllMotors();
  
  Serial.println("EMERGENCY");
  sendStatus();
  
  Serial.println("EMERGENCY STOP ACTIVATED!");
}

void resetSystem() {
  isMoving = false;
  currentState = STATE_READY;
  stopAllMotors();
  
  Serial.println("RESET_COMPLETE");
  sendStatus();
  
  Serial.println("System reset - Emergency cleared");
}

// === Status and Communication ===
void sendStatus() {
  // ✅ Format ที่ ESP32 คาดหวัง: STATUS:B5,S5,L7,READY
  Serial.print("STATUS:B");
  Serial.print(currentPos.backLevel);
  Serial.print(",S");
  Serial.print(currentPos.sideLevel);
  Serial.print(",L");
  Serial.print(currentPos.legLevel);
  Serial.print(",");
  
  switch (currentState) {
    case STATE_READY:
      Serial.println("READY");
      break;
    case STATE_MOVING:
      Serial.println("MOVING");
      break;
    case STATE_EMERGENCY:
      Serial.println("EMERGENCY");
      break;
  }
  
  Serial.flush();
  delay(10);
}

void clearSerialBuffer() {
  while (Serial.available()) {
    Serial.read();
  }
  delay(100);
}

void clearInputBuffer() {
  inputBuffer = "";
}

// === LED Control ===
void updateStatusLED() {
  unsigned long now = millis();
  
  switch (currentState) {
    case STATE_READY:
      digitalWrite(LED_STATUS, HIGH);  // Solid on - ready
      break;
      
    case STATE_MOVING:
      digitalWrite(LED_STATUS, (now % 1000) < 500);  // Slow blink - moving
      break;
      
    case STATE_EMERGENCY:
      digitalWrite(LED_STATUS, (now % 200) < 100);   // Fast blink - emergency
      break;
  }
}
