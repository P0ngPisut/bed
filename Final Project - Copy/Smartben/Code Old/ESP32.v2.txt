// === JSON Buffer ===
// ✅ ขยายจาก 512 -> 2048 (status มี ~33 ฟิลด์ และข้อความไทยใช้ 3 ไบต์/ตัวอักษร เดิมล้นจน JSON ขาด)
StaticJsonDocument<2048> jsonDoc;
char jsonBuffer[2048];

// ส่ง jsonDoc ไป MQTT พร้อมตรวจว่าไม่ล้น (ถ้าล้นจะไม่ส่ง JSON ที่ขาดครึ่งๆ กลางๆ)
bool publishJson(const char* topic) {
  if (jsonDoc.overflowed()) {
    Serial.print("⚠️ JSON overflow, not sent: "); Serial.println(topic);
    return false;
  }
  size_t len = measureJson(jsonDoc);
  if (len >= sizeof(jsonBuffer)) {
    Serial.print("⚠️ JSON too large ("); Serial.print(len); Serial.print(" bytes): "); Serial.println(topic);
    return false;
  }
  serializeJson(jsonDoc, jsonBuffer, sizeof(jsonBuffer));
  return mqttClient.publish(topic, jsonBuffer);
}

void loop() {
  if (!mqttClient.connected()) {
    connectMQTT();
  }
  mqttClient.loop();
  
  if (Serial.available()) {
    handleSerialCommands();
  }
  
  handleArduinoSerial();
  updateLoadCells();
  
  if (autoState.enabled) {
    processEnhancedAutoMode();
  }
  
  unsigned long now = millis();
  
  if (now - lastPingCheck >= PING_INTERVAL) {
    checkArduinoConnection();
    lastPingCheck = now;
  }
  
  if (now - lastHeartbeat >= HEARTBEAT_INTERVAL) {
    sendHeartbeat();
    lastHeartbeat = now;
  }
  
  if (now - lastSensorUpdate >= SENSOR_INTERVAL) {
    readLoadCellData();
    sendSensorData();
    lastSensorUpdate = now;
  }
  
  if (now - lastStatusUpdate >= STATUS_INTERVAL) {
    if (justReceivedCommand && (now - lastCommandTime < COMMAND_COOLDOWN)) {
      // ข้ามการส่ง status
    } else {
      justReceivedCommand = false; 
      sendStatus();
      lastStatusUpdate = now;
    }
  }
  
  if (now - lastLedUpdate >= 500) {
    updateLEDs();
    lastLedUpdate = now;
  }
}

    Serial.println("✅ สะโพก: พร้อมใช้งาน");

}

void connectWiFi() {
  Serial.println("📶 Connecting to WiFi...");
  Serial.print("SSID: "); Serial.println(WIFI_SSID);
  
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 30) {
    delay(500);
    Serial.print(".");
    digitalWrite(LED_WIFI, !digitalRead(LED_WIFI));
    attempts++;
  }
  
  if (WiFi.status() == WL_CONNECTED) {
    digitalWrite(LED_WIFI, HIGH);
    Serial.println();
    Serial.println("✓ WiFi Connected!");
    Serial.print("IP Address: "); Serial.println(WiFi.localIP());
    Serial.print("Signal Strength: "); Serial.print(WiFi.RSSI()); Serial.println(" dBm");
  } else {
    digitalWrite(LED_WIFI, LOW);
    Serial.println();
    Serial.println("❌ WiFi Connection Failed!");
  }
}

void connectMQTT() {
  // ✅ ลองใหม่ทุก 5 วินาทีแบบไม่บล็อก (เดิม delay(5000) ทำให้ไม่ได้อ่านข้อความ EMERGENCY จาก Arduino)
  static unsigned long lastAttempt = 0;
  if (WiFi.status() != WL_CONNECTED) return;
  if (mqttClient.connected()) return;
  if (lastAttempt != 0 && millis() - lastAttempt < 5000) return;
  lastAttempt = millis();

  if (!mqttClient.connected()) {
    Serial.print("🔗 Connecting to HiveMQ Cloud...");
    
    if (mqttClient.connect(CLIENT_ID, MQTT_USER, MQTT_PASS)) {
      digitalWrite(LED_MQTT, HIGH);
      Serial.println(" ✓ Connected!");
      
      mqttClient.subscribe(TOPIC_COMMAND);

      sendStatus();
      
    } else {
      digitalWrite(LED_MQTT, LOW);
      Serial.print(" ❌ Failed! Error code: "); Serial.println(mqttClient.state());
    }
  }
}

void readLoadCellData() {
  // ✅ ระหว่าง Demo ให้ใช้ค่าที่ Demo ตั้งไว้ ไม่อ่านทับ
  if (!autoState.demoRunning) {
    // ✅ ถ้ามีตัวไหนไม่พร้อม จำลองค่าทั้งหมดก่อน แล้วค่อยเขียนทับด้วยค่าจริงของตัวที่พร้อม
    //    (เดิม: ถ้าหลังซ้ายพร้อมแต่หลังขวาเสีย หลังขวาจะเป็น 0 ตลอด -> balance alert ค้าง)
    if (!loadCellStatus.backLeftReady || !loadCellStatus.backRightReady || !loadCellStatus.hipReady) {
      simulateLoadCellData();
    }
    if (loadCellStatus.backLeftReady) {
      sensors.backLeft = max(0.0f, LoadCell_BackLeft.getData() / 1000.0f);
    }
    if (loadCellStatus.backRightReady) {
      sensors.backRight = max(0.0f, LoadCell_BackRight.getData() / 1000.0f);
    }
    if (loadCellStatus.hipReady) {
      sensors.hip = max(0.0f, LoadCell_Hip.getData() / 1000.0f);
    }
  }

  loadCellStatus.lastReadTime = millis();
  
  static unsigned long lastDebugPrint = 0;
  if (millis() - lastDebugPrint > 10000) {
    Serial.print("📊 Load Cell: หลังซ้าย="); Serial.print(sensors.backLeft, 1);
    Serial.print("kg, หลังขวา="); Serial.print(sensors.backRight, 1);
    Serial.print("kg, สะโพก="); Serial.print(sensors.hip, 1); Serial.println("kg");
    
    if (autoState.enabled) {
      Serial.print("🤖 Enhanced Auto: "); Serial.print(autoState.enabled ? "ON" : "OFF");
      Serial.print(", Position: "); Serial.print(autoState.currentPositionName.c_str());
      Serial.print(", Scenario: "); Serial.println(autoState.currentScenario.c_str());
    }
    lastDebugPrint = millis();
  }
}

void simulateLoadCellData() {
  // ✅ Enhanced simulation with side position effects
  float baseBack = 12.0 + (bedPosition.back * 2.5);
  float baseLeg = 15.0 + (bedPosition.leg * 3.0);
  float baseHip = 20.0 - (bedPosition.leg * 1.0);
  
  // ✅ Apply side position effects
  float sideEffect = 0.0;
  if (bedPosition.side < 3) {      // Left side
    sideEffect = (3 - bedPosition.side) * 3.0;  // เพิ่มแรงกดซ้าย
  } else if (bedPosition.side > 7) { // Right side  
    sideEffect = -(bedPosition.side - 7) * 3.0; // เพิ่มแรงกดขวา
  }
  
  // ✅ จำลองทุกครั้ง (เดิมเช็ค scenario == IDLE ทำให้ค่าค้างถาวรหลังเกิด alert ครั้งแรก)
  sensors.backLeft = baseBack + random(-30, 100) / 10.0 + sideEffect;
  sensors.backRight = baseBack + random(-30, 100) / 10.0 - sideEffect;
  sensors.hip = baseHip + random(-50, 150) / 10.0;

  sensors.backLeft = max(0.0f, sensors.backLeft);
  sensors.backRight = max(0.0f, sensors.backRight);
  sensors.hip = max(0.0f, sensors.hip);
}

void loadCalibrationFromStorage() {
  Serial.println("📖 Loading calibration from storage...");
  
  // ✅ ค่า Calibration ที่ได้จากโค้ด calibration
  float calBackLeft = 100.0;    
  float calBackRight = 100.0;   
  float calHip = -110.0;         
  
  if (calBackLeft == -1000.0 || calBackRight == -1000.0 || calHip == -1000.0) {
#if defined(ESP32)
    preferences.begin("smartbed", true);
    
    if (calBackLeft == -1000.0) {
      calBackLeft = preferences.getFloat("cal_left", 1.0);
    }
    if (calBackRight == -1000.0) {
      calBackRight = preferences.getFloat("cal_right", 1.0);
    }
    if (calHip == -1000.0) {
      calHip = preferences.getFloat("cal_hip", 1.0);
    }
    
    preferences.end();
#endif
  }
  
  // ตั้งค่า calibration
  if (loadCellStatus.backLeftReady) {
    LoadCell_BackLeft.setCalFactor(calBackLeft);
    Serial.print("✅ หลังซ้าย: Calibration = "); Serial.println(calBackLeft, 2);
  }
  
  if (loadCellStatus.backRightReady) {
    LoadCell_BackRight.setCalFactor(calBackRight);
    Serial.print("✅ หลังขวา: Calibration = "); Serial.println(calBackRight, 2);
  }
  
  if (loadCellStatus.hipReady) {
    LoadCell_Hip.setCalFactor(calHip);
    Serial.print("✅ สะโพก: Calibration = "); Serial.println(calHip, 2);
  }
  
  if (calBackLeft == 1.0 || calBackRight == 1.0 || calHip == 1.0) {
    Serial.println("💡 ใช้โค้ด Load Cell Test เพื่อหาค่า calibration");
  }
}

void saveCalibrationToStorage(int cellNumber, float value) {
  String cellName, key;
  
  switch (cellNumber) {

    return;

// === Arduino Communication Functions (Same as original) ===
void initializeArduinoConnection() {
  Serial.println("🤖 Initializing Arduino connection...");
  clearArduinoBuffer();
  delay(1000);
  
  for (int i = 0; i < 3; i++) {
    Serial.print("Attempt "); Serial.print(i + 1); Serial.println("/3...");
    
    if (sendArduinoCommandAndWait("PING", 2000)) {
      Serial.println("✓ Arduino responded to PING");
      arduinoConnected = true;
      break;
    }
    delay(1000);
  }
  
  if (arduinoConnected) {
    Serial.println("✓ Arduino connection established");
    sendArduinoCommandAndWait("STATUS", 2000);
  } else {
    Serial.println("⚠️ Arduino connection failed - will retry periodically");
  }
}

// คืนค่า true เฉพาะเมื่อ Arduino ตอบรับคำสั่งจริง (false = ERROR:* หรือ timeout)
bool sendArduinoCommandAndWait(String command, unsigned long timeout) {
  // ✅ ประมวลผลข้อความที่ค้างอยู่ก่อน (เดิม clear ทิ้ง ทำให้ COMPLETE/EMERGENCY หาย)
  handleArduinoSerial();

  Serial.print("📤 Sending to Arduino: "); Serial.println(command);

  pendingCommand = command;
  expectedResponse = expectedReplyFor(command);
  lastCommandOk = false;
  waitingForResponse = true;
  lastCommandSent = millis();

  ArduinoSerial.println(command);
  ArduinoSerial.flush();

  unsigned long startTime = millis();
  while ((millis() - startTime) < timeout) {
    handleArduinoSerial();
    if (!waitingForResponse) {
      return lastCommandOk;
    }
    delay(5);
  }

  waitingForResponse = false;
  return false;
}

void handleArduinoSerial() {
  // ✅ static: เก็บข้อความที่ยังมาไม่ครบบรรทัดไว้ข้ามรอบ loop (9600 baud ส่ง 1 บรรทัดใช้ ~30ms)
  static String response = "";

  while (ArduinoSerial.available()) {
    char c = ArduinoSerial.read();
    if (c == '\n' || c == '\r') {
      if (response.length() > 0) {
        String line = response;
        response = "";
        processArduinoResponse(line);
      }
    } else if (c >= 32 && c <= 126) {
      response += c;
    }

    if (response.length() > 100) {
      Serial.println("⚠️ Arduino response too long - cleared");
      response = "";
    }
  }
}

// ✅ คำตอบที่ถือว่า "สำเร็จ" ของแต่ละคำสั่ง (ERROR:* = ล้มเหลว)
String expectedReplyFor(const String& command) {
  if (command == "PING") return "PONG";
  if (command == "STATUS") return "STATUS:";
  if (command.startsWith("MOVE:") || command == "RESET") return "MOVING";
  if (command == "STOP") return "EMERGENCY";
  return "";
}

void processArduinoResponse(String response) {
  response.trim();
  if (response.length() == 0) return;

  lastResponseReceived = millis();
  arduinoConnected = true;
  connectionAttempts = 0;

  // ✅ HEARTBEAT / echo ไม่ใช่คำตอบของคำสั่ง - แค่ยืนยันว่า Arduino ยังอยู่
  if (response == "HEARTBEAT" || response.startsWith("RECEIVED:")) {
    return;
  }

  Serial.print("📥 Arduino: "); Serial.println(response);

  // ✅ เคลียร์สถานะรอ เฉพาะเมื่อได้คำตอบที่ตรงกับคำสั่งที่ส่งไป
  if (waitingForResponse) {
    if (response.startsWith("ERROR:")) {
      lastCommandOk = false;
      waitingForResponse = false;
    } else if (expectedResponse.length() == 0 || response.startsWith(expectedResponse)) {
      lastCommandOk = true;
      waitingForResponse = false;
    }
  }

  // เก็บสถานะเดิมไว้ เพื่อส่ง status ไป MQTT เฉพาะตอนที่มีการเปลี่ยนแปลง
  String prevBedStatus = bedStatus;
  String prevArduinoStatus = arduinoStatus;
  int prevBack = bedPosition.back, prevSide = bedPosition.side, prevLeg = bedPosition.leg;

  if (response.startsWith("STATUS:")) {
    parseArduinoStatus(response.substring(7));
    arduinoStatus = "CONNECTED";
  } else if (response == "PONG") {
    Serial.println("✓ Arduino PING successful");
    arduinoStatus = "CONNECTED";
  } else if (response == "MOVING") {
    bedStatus = "MOVING";
    sendLog("Arduino: Movement started");
  } else if (response == "MANUAL_COMPLETE") {
    bedStatus = "READY";
    sendLog("Arduino: Manual actuator correction completed");
  } else if (response == "COMPLETE") {
    // Arduino ส่ง STATUS ตามมาเองหลัง COMPLETE ไม่ต้องสั่ง STATUS ซ้อน (กันการเรียกซ้อนกัน)
    bedStatus = "READY";
    sendLog("Arduino: Movement completed");
  } else if (response == "EMERGENCY") {
    bedStatus = "EMERGENCY";
    autoState.enabled = false; 
    sendLog("Arduino: EMERGENCY STOP!");
  } else if (response == "RESET_COMPLETE") {
    bedStatus = "READY";
    sendLog("Arduino: System reset");
  } else if (response.startsWith("Back:") || response.startsWith("Side:") || response.startsWith("Leg:")) {
    sendLog("Arduino: " + response);
  } else if (response.startsWith("ERROR:")) {
    sendLog("Arduino Error: " + response);
  } else if (response == "ARDUINO_READY") {
    arduinoConnected = true;
    arduinoStatus = "READY";
    sendLog("Arduino: System ready");
  } else {
    sendLog("Arduino: " + response);
  }

  // ✅ ส่ง status เฉพาะเมื่อมีอะไรเปลี่ยน (เดิมส่งทุกบรรทัด ทำให้ MQTT ท่วม)
  if (bedStatus != prevBedStatus || arduinoStatus != prevArduinoStatus ||
      bedPosition.back != prevBack || bedPosition.side != prevSide || bedPosition.leg != prevLeg) {
    sendStatus();
  }
}

void parseArduinoStatus(String statusData) {
  int commaIndex1 = statusData.indexOf(',');
  int commaIndex2 = statusData.indexOf(',', commaIndex1 + 1);
  int commaIndex3 = statusData.indexOf(',', commaIndex2 + 1);
  
  if (commaIndex1 > 0 && commaIndex2 > 0) {
    String backStr = statusData.substring(1, commaIndex1); 
    String sideStr = statusData.substring(commaIndex1 + 2, commaIndex2); 
    String legStr = statusData.substring(commaIndex2 + 2, commaIndex3 > 0 ? commaIndex3 : statusData.length()); 
    
    int newBack = backStr.toInt();
    int newSide = sideStr.toInt();
    int newLeg = legStr.toInt();
    
    // ✅ ตำแหน่งจาก Arduino คือตำแหน่งจริง - ใช้ค่านี้เสมอ (ESP32 ไม่เดาตำแหน่งล่วงหน้าแล้ว)
    bedPosition.back = constrain(newBack, -10, 10);
    bedPosition.side = constrain(newSide, -10, 10);
    bedPosition.leg = constrain(newLeg, -10, 10);

    if (commaIndex3 > 0) {
      String status = statusData.substring(commaIndex3 + 1);
      status.trim();

      if (status == "EMERGENCY") {
        bedStatus = "EMERGENCY";
      } else if (status == "MOVING") {
        bedStatus = "MOVING";
      } else if (status == "READY") {
        bedStatus = "READY";
        autoState.currentPositionName = getEnhancedCurrentPositionName();
      }
      
      arduinoStatus = status;
    }
    
    Serial.print("📊 Arduino Status: "); Serial.print(autoState.currentPositionName.c_str());
    Serial.print(" ("); Serial.print(newBack); Serial.print(","); Serial.print(newSide); Serial.print(","); Serial.print(newLeg);
    Serial.print("), Status="); Serial.println(arduinoStatus.c_str());
  }
}

void checkArduinoConnection() {
  if (!waitingForResponse) {
    if (arduinoConnected) {
      // ✅ Arduino ส่ง HEARTBEAT ทุก 1 วินาที ถ้ายังได้รับอยู่ไม่ต้อง PING (ไม่ต้องบล็อก loop)
      if (millis() - lastResponseReceived < PING_INTERVAL) return;
      if (!sendArduinoCommandAndWait("PING", 1000)) {
        handleArduinoTimeout();
      }
    } else {
      Serial.print("🔄 Attempting to reconnect to Arduino... ");
      if (sendArduinoCommandAndWait("PING", 2000)) {
        Serial.println("✓ Reconnected!");
        arduinoConnected = true;
        sendLog("Arduino reconnected");
      } else {
        Serial.println("❌ Still disconnected");
        connectionAttempts++;
        if (connectionAttempts >= maxConnectionAttempts) {
          Serial.println("⚠️ Max connection attempts reached");
        }
      }
    }
  }
}

void handleArduinoTimeout() {
  waitingForResponse = false;
  connectionAttempts++;
  
  if (connectionAttempts >= maxConnectionAttempts) {
    arduinoConnected = false;
    arduinoStatus = "TIMEOUT";
    sendLog("Arduino communication timeout - marked as disconnected");
  } else {
    sendLog("Arduino timeout - retrying...");
  }
}

void clearArduinoBuffer() {
  while (ArduinoSerial.available()) {
    ArduinoSerial.read();
  }
  delay(50);
}

// === MQTT Callback & Command Processing ===
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (unsigned int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  
  Serial.println(); Serial.println("📨 Enhanced Dashboard Command:");
  Serial.print("Topic: "); Serial.println(topic);
  Serial.print("Message: "); Serial.println(message);
  
  // ✅ ใช้ document แยกสำหรับคำสั่ง - jsonDoc ตัวกลางถูก sendLog()/sendStatus() ล้างทิ้งระหว่างประมวลผล
  StaticJsonDocument<512> cmdDoc;
  DeserializationError error = deserializeJson(cmdDoc, message);
  if (error) {
    Serial.print("❌ JSON parsing failed: "); Serial.println(error.c_str());
    return;
  }

  const char* cmd = cmdDoc["command"];
  if (!cmd) {
    Serial.println("❌ No command in message");
    return;
  }

  String command = String(cmd);
  command.toUpperCase();

  Serial.print("🎮 Processing Enhanced command: "); Serial.println(command);
  processEnhancedCommand(command, cmdDoc);
}

void processEnhancedCommand(String command, JsonDocument& data) {
  justReceivedCommand = true;
  lastCommandTime = millis();
  
  if (command == "STATUS") {
    Serial.println("📊 Dashboard requested status");
    if (arduinoConnected) {
      sendArduinoCommandAndWait("STATUS", 2000);
    }
    sendStatus();
    sendSensorData();
  }
  else if (command == "MOVE") {
    if (!arduinoConnected) {
      Serial.println("❌ Cannot move - Arduino not connected");
      sendLog("Move failed: Arduino not connected");
      return;
    }
    
    if (bedStatus == "EMERGENCY") {
      sendLog("Move failed: Emergency mode active");
      return;
    }
    
    int back = data["back"] | bedPosition.back;
    int side = data["side"] | bedPosition.side;
    int leg = data["leg"] | bedPosition.leg;
    
    back = constrain(back, -10, 10);
    side = constrain(side, -10, 10);
    leg = constrain(leg, -10, 10);
    
    String posName = positionNameFor(back, side, leg);  // ✅ ชื่อของตำแหน่งใหม่ (เดิมใช้ตำแหน่งเก่า)
    executeEnhancedPositionChange(back, side, leg, "Manual Control", posName);
  }
  else if (command == "PRESET") {
    if (!arduinoConnected) {
      sendLog("Preset failed: Arduino not connected");
      return;
    }
    
    const char* preset = data["preset"];
    if (!preset) return;
    
    int back, side, leg;
    String posName;
    
    if (strcmp(preset, "FLAT") == 0) {
      back = 0; side = 5; leg = 5; posName = "FLAT";
    } else if (strcmp(preset, "SIT") == 0) {
      back = 7; side = 5; leg = 2; posName = "SIT";
    } else if (strcmp(preset, "SLEEP") == 0) {
      back = 2; side = 5; leg = 1; posName = "SLEEP";
    } else if (strcmp(preset, "LEFT_SIDE") == 0) {
      back = 0; side = 1; leg = 5; posName = "LEFT_SIDE";  // ✅ New preset
    } else if (strcmp(preset, "RIGHT_SIDE") == 0) {
      back = 0; side = 9; leg = 5; posName = "RIGHT_SIDE"; // ✅ New preset
    } else {
      return;
    }
    
    executeEnhancedPositionChange(back, side, leg, "Preset: " + String(preset), posName);
  }
  else if (command == "AUTO") {
    bool enable = data["enable"] | false;
    autoState.enabled = enable;
    
    if (enable) {
      // รีเซ็ตตัวจับเวลาทั้งหมด
      autoState.backLeftHighStart = 0;
      autoState.backRightHighStart = 0;
      autoState.hipHighStart = 0;
      autoState.balanceIssueStart = 0;
      
      autoState.emergencyMode = false;
      autoState.currentScenario = "IDLE";
      
      // รีเซ็ต Alert States
      autoState.backLeftAlert = false;
      autoState.backRightAlert = false;
      autoState.hipAlert = false;
      autoState.balanceAlert = false;
      
      sendLog("Enhanced Auto mode enabled - weight-based monitoring started");
    } else {
      autoState.demoRunning = false;
      autoState.emergencyMode = false;
      autoState.currentScenario = "IDLE";
      sendLog("Enhanced Auto mode disabled");
    }
    
    Serial.print("⚙️ Enhanced Auto mode: "); Serial.println(enable ? "ENABLED" : "DISABLED");
    sendStatus();
  }
  else if (command == "DEMO") {
    String demoAction = data["action"] | "start";
    
    if (demoAction == "start") {
      autoState.demoRunning = true;
      autoState.enabled = true; 
      autoState.currentScenario = "DEMO_STARTING";
      sendLog("🎭 Enhanced Demo mode started!");
      sendAutoAlert("DEMO_STARTED", "เริ่ม Enhanced Demo การปรับตำแหน่งอัตโนมัติ", "INFO");
    } else if (demoAction == "stop") {
      autoState.demoRunning = false;
      autoState.currentScenario = "IDLE";
      sendLog("🎭 Enhanced Demo mode stopped");
      sendAutoAlert("DEMO_STOPPED", "หยุด Enhanced Demo", "INFO");
    }
    
    Serial.print("🎭 Enhanced Demo mode: "); Serial.println(autoState.demoRunning ? "STARTED" : "STOPPED");
    sendStatus();
  }
  else if (command == "AUTO_CONFIG") {
    // ✅ Enhanced Auto Configuration
    if (data.containsKey("pressure_seconds")) {
      autoConfig.pressureCheckSeconds = data["pressure_seconds"] | autoConfig.pressureCheckSeconds;
    }
    if (data.containsKey("balance_seconds")) {
      autoConfig.balanceCheckSeconds = data["balance_seconds"] | autoConfig.balanceCheckSeconds;
    }
    if (data.containsKey("hip_seconds")) {
      autoConfig.hipPressureSeconds = data["hip_seconds"] | autoConfig.hipPressureSeconds;
    }
    if (data.containsKey("pressure_threshold")) {
      autoConfig.highPressureThreshold = data["pressure_threshold"] | autoConfig.highPressureThreshold;
    }
    if (data.containsKey("balance_threshold")) {
      autoConfig.balanceThreshold = data["balance_threshold"] | autoConfig.balanceThreshold;
    }
    
    Serial.println("⚙️ Enhanced Auto Config Updated:");
    Serial.print("   Pressure: "); Serial.print(autoConfig.pressureCheckSeconds); Serial.println("s");
    Serial.print("   Balance: "); Serial.print(autoConfig.balanceCheckSeconds); Serial.println("s");
    Serial.print("   Hip: "); Serial.print(autoConfig.hipPressureSeconds); Serial.println("s");
    
    sendLog("Enhanced Auto configuration updated");
    sendStatus();
  }
  else if (command == "STOP") {
    Serial.println("🛑 EMERGENCY STOP from Dashboard!");
    
    if (arduinoConnected) {
      sendArduinoCommandAndWait("STOP", 2000);
    }
    
    bedStatus = "EMERGENCY";
    autoState.enabled = false;
    autoState.demoRunning = false;
    
    sendLog("EMERGENCY STOP activated from Dashboard!");
    sendStatus();
  }
  else if (command == "RESET") {
    Serial.println("🔄 System Reset from Dashboard");
    
    if (arduinoConnected) {
      sendArduinoCommandAndWait("RESET", 2000);
    }
    
    autoState.enabled = false;
    autoState.demoRunning = false;
    autoState.emergencyMode = false;
    autoState.currentScenario = "IDLE";
    autoState.currentPositionName = "FLAT";
    
    // รีเซ็ตตัวจับเวลา
    autoState.backLeftHighStart = 0;
    autoState.backRightHighStart = 0;
    autoState.hipHighStart = 0;
    autoState.balanceIssueStart = 0;
    
    sendLog("Enhanced System reset from Dashboard - returning to FLAT");
  }
  else {
    Serial.print("❓ Unknown command: "); Serial.println(command);
  }
}

// === Data Sending Functions ===
void sendStatus() {
  if (!mqttClient.connected()) return;
  
  jsonDoc.clear();
  jsonDoc["timestamp"] = millis();
  jsonDoc["device_id"] = CLIENT_ID;
  jsonDoc["uptime"] = (millis() - systemStartTime) / 1000;
  
  JsonObject position = jsonDoc.createNestedObject("position");
  position["back"] = bedPosition.back;
  position["side"] = bedPosition.side;
  position["leg"] = bedPosition.leg;
  position["name"] = autoState.currentPositionName;
  
  JsonObject target = jsonDoc.createNestedObject("target");
  target["back"] = targetPosition.back;
  target["side"] = targetPosition.side;
  target["leg"] = targetPosition.leg;
  
  jsonDoc["mode"] = autoState.enabled ? "AUTO" : "MANUAL";
  jsonDoc["status"] = bedStatus;
  jsonDoc["arduino_connected"] = arduinoConnected;
  jsonDoc["arduino_status"] = arduinoStatus;
  jsonDoc["connection_attempts"] = connectionAttempts;
  jsonDoc["wifi_rssi"] = WiFi.RSSI();
  jsonDoc["free_heap"] = ESP.getFreeHeap();
  jsonDoc["loadcells_ready"] = loadCellsReady;
  
  // ✅ Enhanced Auto Status
  JsonObject autoStatus = jsonDoc.createNestedObject("auto_mode");
  autoStatus["enabled"] = autoState.enabled;
  autoStatus["demo_running"] = autoState.demoRunning;
  autoStatus["emergency_mode"] = autoState.emergencyMode;
  autoStatus["current_scenario"] = autoState.currentScenario;
  autoStatus["back_left_alert"] = autoState.backLeftAlert;
  autoStatus["back_right_alert"] = autoState.backRightAlert;
  autoStatus["hip_alert"] = autoState.hipAlert;
  autoStatus["balance_alert"] = autoState.balanceAlert;
  
  // ✅ Enhanced Configuration
  JsonObject autoConfigObj = jsonDoc.createNestedObject("auto_config");
  autoConfigObj["pressure_seconds"] = autoConfig.pressureCheckSeconds;
  autoConfigObj["balance_seconds"] = autoConfig.balanceCheckSeconds;
  autoConfigObj["hip_seconds"] = autoConfig.hipPressureSeconds;
  autoConfigObj["pressure_threshold"] = autoConfig.highPressureThreshold;
  autoConfigObj["balance_threshold"] = autoConfig.balanceThreshold;
  
  publishJson(TOPIC_STATUS);
  
  Serial.println("📤 Enhanced Status sent to Dashboard");
}

void sendSensorData() {
  if (!mqttClient.connected()) return;
  
  jsonDoc.clear();
  jsonDoc["timestamp"] = millis();
  
  JsonObject sensorObj = jsonDoc.createNestedObject("sensors");
  sensorObj["back_left"] = sensors.backLeft;
  sensorObj["back_right"] = sensors.backRight;
  sensorObj["hip"] = sensors.hip;
  sensorObj["total"] = sensors.backLeft + sensors.backRight + sensors.hip;
  
  JsonObject pressure = jsonDoc.createNestedObject("pressure_alert");
  pressure["back_left"] = (sensors.backLeft > autoConfig.highPressureThreshold);
  pressure["back_right"] = (sensors.backRight > autoConfig.highPressureThreshold);
  pressure["hip"] = (sensors.hip > autoConfig.highPressureThreshold);
  
  float diff = abs(sensors.backLeft - sensors.backRight);
  jsonDoc["balance_diff"] = diff;
  jsonDoc["unbalanced"] = (diff > autoConfig.balanceThreshold);
  
  JsonObject loadcell_status = jsonDoc.createNestedObject("loadcell_status");
  loadcell_status["back_left_ready"] = loadCellStatus.backLeftReady;
  loadcell_status["back_right_ready"] = loadCellStatus.backRightReady;
  loadcell_status["hip_ready"] = loadCellStatus.hipReady;
  loadcell_status["all_ready"] = loadCellsReady;
  
  JsonObject thresholds = jsonDoc.createNestedObject("auto_thresholds");
  thresholds["high_pressure"] = autoConfig.highPressureThreshold;
  thresholds["balance"] = autoConfig.balanceThreshold;
  
  publishJson(TOPIC_SENSOR);
}

// === Utility Functions ===
void updateLEDs() {
  if (systemReady && WiFi.status() == WL_CONNECTED && mqttClient.connected()) {
    digitalWrite(LED_STATUS, HIGH);
  } else {
    digitalWrite(LED_STATUS, millis() % 1000 < 500);
  }
  
  digitalWrite(LED_WIFI, WiFi.status() == WL_CONNECTED);
  digitalWrite(LED_MQTT, mqttClient.connected());
  
  if (arduinoConnected && (loadCellsReady || !autoConfig.demoMode)) {
    digitalWrite(LED_ARDUINO, HIGH);
  } else {
    digitalWrite(LED_ARDUINO, millis() % 2000 < 1000);
  }
}

// === Serial Commands ===
void handleSerialCommands() {
  String cmd = Serial.readStringUntil('\n');
  cmd.trim();
  
  if (cmd.length() == 0) return;
  
  Serial.println(); Serial.print("💻 Enhanced Serial Command: "); Serial.println(cmd);
  
  char firstChar = cmd.charAt(0);
  
  switch (firstChar) {
    case 't':
    case 'T':
      tareAllLoadCells();
      break;
      
    case 'r':
    case 'R':
      calibrationMenu();
      break;
      
    case 'i':
    case 'I':
      printEnhancedLoadCellInfo();
      break;
      
    case 'h':
    case 'H':
      printEnhancedHelp();
      break;
      
    default:
      cmd.toUpperCase();
      if (cmd == "AUTO ON") {
        autoState.enabled = true;
        Serial.println("🤖 Enhanced Auto mode enabled via serial");
        sendLog("Enhanced Auto mode enabled via serial");
      }
      else if (cmd == "AUTO OFF") {
        autoState.enabled = false;
        autoState.demoRunning = false;
        Serial.println("🤖 Enhanced Auto mode disabled via serial");
        sendLog("Enhanced Auto mode disabled via serial");
      }
      else if (cmd == "DEMO START") {
        autoState.demoRunning = true;
        autoState.enabled = true;
        Serial.println("🎭 Enhanced Demo started via serial");
        sendLog("Enhanced Demo started via serial");
      }
      else if (cmd == "DEMO STOP") {
        autoState.demoRunning = false;
        Serial.println("🎭 Enhanced Demo stopped via serial");
        sendLog("Enhanced Demo stopped via serial");
      }
      else if (cmd == "STATUS" || cmd == "?") {
        printEnhancedSystemStatus();
      }
      else if (cmd == "CONFIG") {
        printEnhancedAutoConfig();
      }
      else if (cmd == "ARDUINO") {
        testArduinoConnection();
      }
      else if (cmd == "RECONNECT") {
        reconnectArduino();
      }
      else if (cmd == "TEST") {
        testEnhancedDashboardConnection();
      }
      else if (cmd.startsWith("MOVE ")) {
        String params = cmd.substring(5);
        processEnhancedManualMove(params);
      }
      else if (cmd.startsWith("PRESET ")) {
        String preset = cmd.substring(7);
        processEnhancedPresetMove(preset);
      }
      else {
        Serial.println("❓ Unknown command. Type 'h' for available commands.");
      }
      break;
  }
}

void processEnhancedManualMove(String params) {
  int comma1 = params.indexOf(',');
  int comma2 = params.lastIndexOf(',');
  
  if (comma1 > 0 && comma2 > comma1) {
    int back = params.substring(0, comma1).toInt();
    int side = params.substring(comma1 + 1, comma2).toInt();
    int leg = params.substring(comma2 + 1).toInt();
    
    back = constrain(back, -10, 10);
    side = constrain(side, -10, 10);
    leg = constrain(leg, -10, 10);
    
    String posName = positionNameFor(back, side, leg);  // ✅ ชื่อของตำแหน่งใหม่ (เดิมใช้ตำแหน่งเก่า)
    executeEnhancedPositionChange(back, side, leg, "Manual Serial Command", posName);
    Serial.print("✓ Enhanced manual move: "); Serial.print(back); Serial.print(","); Serial.print(side); Serial.print(","); Serial.println(leg);
  } else {
    Serial.println("❌ Invalid format. Use: MOVE back,side,leg (e.g., MOVE 0,5,5)");
  }
}

void processEnhancedPresetMove(String preset) {
  preset.toUpperCase();
  
  int back, side, leg;
  String posName;
  
  if (preset == "FLAT") {
    back = 0; side = 5; leg = 5; posName = "FLAT";
  } else if (preset == "SIT") {
    back = 7; side = 5; leg = 2; posName = "SIT";
  } else if (preset == "SLEEP") {
    back = 2; side = 5; leg = 1; posName = "SLEEP";
  } else if (preset == "LEFT_SIDE" || preset == "LEFT") {
    back = 0; side = 1; leg = 5; posName = "LEFT_SIDE";
  } else if (preset == "RIGHT_SIDE" || preset == "RIGHT") {
    back = 0; side = 9; leg = 5; posName = "RIGHT_SIDE";
  } else {
    Serial.println("❌ Unknown preset. Available: FLAT, SIT, SLEEP, LEFT_SIDE, RIGHT_SIDE");
    return;
  }
  
  executeEnhancedPositionChange(back, side, leg, "Serial Preset: " + preset, posName);
  Serial.print("✓ Enhanced preset: "); Serial.print(preset.c_str()); Serial.print(" -> "); Serial.print(posName.c_str());
  Serial.print(" ("); Serial.print(back); Serial.print(","); Serial.print(side); Serial.print(","); Serial.print(leg); Serial.println(")");
}
